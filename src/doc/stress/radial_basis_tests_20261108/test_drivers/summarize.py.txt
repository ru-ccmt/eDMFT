"""Collect the completed radial tests; archive data, never change source."""
from pathlib import Path
import json,re,hashlib,shutil

ROOT=Path(__file__).resolve().parent
OUT=ROOT/'deliverables'
OUT.mkdir(exist_ok=True)
summary={}
def number(path,tag):
    text=path.read_text()
    return float(re.findall(r'^'+re.escape(tag)+r'\s+([-+0-9.Ee]+)',text,re.M)[-1])
for case in ['SrVO3','FeSe']:
    d=ROOT/(case+'_DFT');m=ROOT/(case+'_DMFT')
    fresh=json.loads((d/'fresh_dft_stress.json').read_text())
    eref=json.loads((d/'energy_reference_test.json').read_text())
    ds=json.loads((d/'step_check.json').read_text());ms=json.loads((m/'step_check.json').read_text())
    mc=json.loads((m/'closure.json').read_text())
    b=fresh['mean_GPa'];num=fresh['numeric_GPa'];rad=eref['results'][-1]['fixed_weights_GPa']
    old=2.690161085247642 if case=='SrVO3' else -1.118667459278174
    mn=2.933836407 if case=='SrVO3' else -1.0491512951
    mr=ms['half_step_GPa']
    rawdelta=number(m/'radial_projector.out',':DMFT_STRESS_DIAG_EXPLICIT_PROJECTOR_MEAN_GPA')-number(m/'radial_contraction.out',':DMFT_STRESS_DIAG_EXPLICIT_PROJECTOR_MEAN_GPA')
    legacy=number(m/'radial_contraction.out',':DMFT_STRESS_RADIAL_FD_MEAN_DK_GPA')-rawdelta
    summary[case]={
      'DFT':{'analytic_fresh_GPa':b,'numeric_saved_GPa':num,'before_residual_GPa':b-num,
             'radial_fixed_Eref_GPa':ds['half_fixed_weights_GPa'],
             'radial_with_Eref_GPa':rad,'radial_with_Eref_fullstep_GPa':eref['results'][0]['fixed_weights_GPa'],
             'diagnostic_sum_GPa':b+rad,'diagnostic_sum_minus_numeric_GPa':b+rad-num,
             'legacy_radial_fixed_Eref_GPa':number(d/'small_diag'/(case+'.outputdmf2'),':DMFT_STRESS_RADIAL_FD_MEAN_DK_GPA')},
      'eDMFT':{'analytic_with_previous_potential_response_GPa':old,'numeric_saved_GPa':mn,
               'before_residual_GPa':old-mn,'radial_fullstep_GPa':mc['basis_free_energy_fd_GPa'],
               'radial_halfstep_GPa':mr,'diagnostic_sum_GPa':old+mr,'diagnostic_sum_minus_numeric_GPa':old+mr-mn,
               'legacy_HO_plus_projector_GPa':legacy,'legacy_minus_free_energy_GPa':legacy-mr},
      'zero_DFT_energy_error_Ry':ds['zero_max_energy_error_Ry'],
      'zero_DMFT_VSUM_error_Ry':ms['zero_vsum_error_Ry']}
(OUT/'SUMMARY.json').write_text(json.dumps(summary,indent=2)+'\n')
