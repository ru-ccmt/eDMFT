"""DFT radial response including the actual IN1 Fermi-reference change."""
from run_tests import *
from step_check import interpolate

def run(case):
    root=ROOT/(case+'_DFT');c=root/'center';summary=json.loads((root/'summary.json').read_text())
    is_c=case=='SrVO3';suf='in1c' if is_c else 'in1';in2=case+('.in2c' if is_c else '.in2')
    orig=Path('/Users/haule/projects/materials/SrVO3') if is_c else Path('/Users/haule/projects/materials/FeSe_again/FeSe_stress')
    heads=[]
    for label in ['m001','p001']:
        name='DFT_'+label if is_c else 'DFT_RELA_'+label
        heads.append((orig/name/(case+'.'+suf)).read_text().splitlines()[0])
    ef=lambda x:float(re.search(r'EF=\s*([-+0-9.Ee]+)',x).group(1))
    ec=ef((c/(case+'.'+suf)).read_text().splitlines()[0]);em,ep=map(ef,heads)
    def weight(path):
        result=[]
        for l in path.read_text().splitlines():
            a=l.split()
            if len(a)==4 and a[0].isdigit() and a[1].isdigit():result.append(list(map(float,a)))
        return np.array(result)
    wc=weight(c/(case+'.weight'));results=[]
    for scale in [1.,.5]:
        energies=[];sums=[]
        for sign,eref in zip(['minus','plus'],[em,ep]):
            d=root/(f'eref_{scale:g}_'+sign);clone(c,d,case)
            shutil.copy2(c/'lapw1.fixgvec',d/'lapw1.fixgvec');(d/'lapw1.fixgvec.read').touch()
            interpolate(c/(case+'.vsp'),root/sign/'basis.vsp',d/'basis.vsp',scale)
            p=d/(case+'.'+suf);a=p.read_text().splitlines();e=ec+scale*(eref-ec);a[0]=f'WFFIL EF= {e:.15f} Old {e:.15f}';p.write_text('\n'.join(a)+'\n')
            p=d/in2;a=p.read_text().splitlines();a[0]='TOT '+a[0][4:];p.write_text('\n'.join(a)+'\n')
            execute(d,'lapw1',[str(WIEN/('lapw1c' if is_c else 'lapw1')),'lapw1.def'],{'WIEN_LAPW1_BASIS_VSP':str(d/'basis.vsp')})
            shutil.copy2(d/'basis.vsp',d/(case+'.vsp'))
            execute(d,'lapw2',[str(WIEN/('lapw2c' if is_c else 'lapw2')),'lapw2.def'])
            energies.append(weight(d/(case+'.weight'))[:,2]);sums.append(float(re.findall(r'^:SUM.*?=\s*([-+0-9.Ee]+)',(d/(case+'.scf2')).read_text(),re.M)[-1]))
        factor=14710.507641078/(2*scale*summary['step']*3*summary['volume'])
        results.append({'scale':scale,'fixed_weights_GPa':float(np.dot(wc[:,3],energies[1]-energies[0])*factor),'sum_fd_GPa':(sums[1]-sums[0])*factor})
    result={'ef_center':ec,'ef_minus':em,'ef_plus':ep,'results':results}
    (root/'energy_reference_test.json').write_text(json.dumps(result,indent=2)+'\n');print(result,flush=True)

if __name__=='__main__':run(sys.argv[1])
